\section{Notation}
\label{spec:notation}

\begin{tabular}{@{}ll@{}}
\toprule
Symbol & Meaning \\
\midrule
$W_u$, $\pi_u$, $\sgn_u$ & mode, propensity and request direction of unit $u$ \\
$L(\accept) = 1$, $L(\reject) = 0$ & mode indicator \\
$v_u$, $w_u$ & assignment score and weight of unit $u$ \\
$S$, $\sigma^2$, $z$ & statistic, design variance, normalised statistic \\
$B$, $z_b$, $p$, $p_d$, $o$ & redraws, redrawn statistic, two-sided and one-sided p-values, orientation \\
$\Hnull(f, r, k)$, $\Hset$ & sharp null of a hypothesis; set of true sharp nulls \\
$\Gpast{u-1}$, $\Fpast{u-1}$ & history before unit $u$ draws its mode; natural history \\
$\taupi$, $\tauAA$ & policy-averaged assignment effect; knockout effect \\
$m$, $q$, $\omega_j$, $d_j$, $\cm$ & hypotheses, FDR level, weights, prior directions, $\sum_{i=1}^m 1/i$ \\
$\pimin$ & lower bound of the propensities \\
$\hat\beta$, $e_u$ & design slope, centred outcome \\
$\varepsilon(f, d, k)$, $G_k$, $N(f, d)$ & gate tolerance, per-episode guard total, units per class \\
$\delta_k$ & materiality threshold of the ground truth \\
$R$, $V$, $\VAA$, $\Vref$ & recovery, protected-slice violations, accept-all and reference values \\
\bottomrule
\end{tabular}
