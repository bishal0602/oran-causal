\subsection{Fresh draws}
\label{spec:thm.fresh}

\begin{lemma}[Fresh draw]
Let $(U_\xi)_{\xi \in K}$, $K$ countable, be jointly independent uniforms on $(0,1)$ and independent of $\Gpast{0}$. Let $\xi_u$ and $\pi_u \in [0,1]$ be $\Gpast{u-1}$-measurable, with $\xi_u$ different from the earlier keys $\xi_1, \dots, \xi_{u-1}$, and let the keys be recursively predictable (each $\xi_i$ is determined by $\Gpast{i-1}$). Then $U_{\xi_u}$ is uniform on $(0,1)$ and independent of $\Gpast{u-1}$, hence $\Prob(W_u = \accept \mid \Gpast{u-1}) = \pi_u$ for $W_u = \accept \iff U_{\xi_u} < \pi_u$.
\end{lemma}

\noindent Only the current key needs to be fresh; later keys may repeat it.\claim{THM-FRESH}

\subsection{Asymptotic validity of PMRT}
\label{spec:thm.pmrt}

\begin{proposition}[Asymptotic validity]
Under (A1)--(A3), $(S_j)$ with $S_j = \sum_{u \le j} v_u w_u$ is a martingale with respect to $(\Gpast{j})$ with predictable quadratic variation $\sigma^2$. Under (A1)--(A4), $z = S/\sigma \Rightarrow \Normal(0,1)$, the conditional distribution of $z_b$ given the data converges to $\Phi$ uniformly in probability, and
\begin{equation*}
\limsup_{n \to \infty} \Prob(p \le t) \le \frac{\floor{t\,(B + 1)}}{B + 1} \le t \qquad \text{for every } t \in [0,1].
\end{equation*}
\end{proposition}

\noindent The martingale property needs $w_u$ square integrable (Lean \texttt{Asymptotic.pmrt\_martingale}, a draft not yet part of the imported library); the central limit theorems for the statistic and its redraws rely on (A4) with $\sigma^2/s_n^2 \to \eta^2 \in (0,\infty)$ and are not formalised. The conclusion is not exact for finite $n$; see Section~\ref{spec:thm.exact} for the case with fixed design.\claim{THM-PMRT-ASY}

\subsection{Exact case}
\label{spec:thm.exact}

\begin{lemma}[Exact validity]
Assume (A1) and (A2) and that the realised units, their order, $\sgn_u$, $\pi_u$ and $w_u$ are $\Gpast{0}$-measurable. Let the redraws use fresh keys. Then for every $t \in [0,1]$, every number of units $n$ and every number of redraws $B \ge 0$ (including ties, zero variance and an empty unit set),
\begin{equation*}
\Prob(p \le t \mid \Gpast{0}) \le t,
\end{equation*}
and likewise for $p_d$ with a direction $d$ fixed and an orientation $o$ that is $\Gpast{0}$-measurable.
\end{lemma}

\noindent The observed statistic and the $B$ redrawn statistics are exchangeable given $\Gpast{0}$ under $\Hnull$, and the plus-one p-value of an exchangeable vector is super-uniform.\claim{THM-EXACT}

\subsection{Re-proposal counterexample}
\label{spec:thm.counterex}

\begin{proposition}[Re-proposal breaks finite-sample validity]
In a two-unit design in which a rejected unit is proposed again ($\pi_u = 1/2$, $\sgn_u = o = +1$, $w_1 = 1$, $w_2 = a > 1$), the plus-one p-value satisfies $\Prob(p_{+1} \le t) \to 3/4 > t$ as $B \to \infty$ for every fixed $t \in (1/2, 3/4)$ (Lean \texttt{finite\_redraw\_rejection\_limit\_between\_half\_and\_three\_quarters}; the limit is $3/4$ for every $t \in (1/2, 1)$, \texttt{finite\_redraw\_rejection\_limit\_above\_half}). At the boundary $t = 1/2$ the exact-tail test rejects with probability $3/4$ (\texttt{exact\_tail\_rejection}) while the plus-one p-value gives $3/8$ in the limit (\texttt{finite\_redraw\_rejection\_limit}, \texttt{finite\_redraw\_rejection\_formula}).
\end{proposition}

\noindent Thus re-proposal alone breaks finite-sample validity of the plus-one redraw p-value.\claim{THM-COUNTEREX}

\subsection{Weighted BY under arbitrary dependence}
\label{spec:thm.byfin}

Let $\cm = \sum_{i=1}^m 1/i$ and $t_{j,i} = q\,\omega_j\, i/(m\,\cm)$, $t_{j,0} = 0$.

\begin{theorem}[Finite weighted BY]
Let $m \ge 1$, $q \ge 0$ and $\omega_j \ge 0$ deterministic, the p-values $p_j$ and the rejection set $\mathcal{S}$ measurable, and suppose every $j \in \mathcal{S}$ satisfies $p_j \le t_{j,\abs{\mathcal{S}}}$. Let $\rho = \abs{\mathcal{S}}$. For a null $j$ with $F_j(t) = \Prob(p_j \le t)$ and any joint law,
\begin{equation*}
\Ex\Bigl[\frac{\Ind\{p_j \le t_{j,\rho}\}}{\max(\rho, 1)}\Bigr]
\le \sum_{i=1}^{m-1} F_j(t_{j,i}) \Bigl(\frac{1}{i} - \frac{1}{i+1}\Bigr) + \frac{F_j(t_{j,m})}{m}.
\end{equation*}
Hence if every null p-value has $F_j(t_{j,i}) \le t_{j,i} + \delta$ for $i = 1, \dots, m$,
\begin{equation*}
\FDR \le \frac{q}{m}\sum_{j \in \Hset} \omega_j + \abs{\Hset}\,\delta .
\end{equation*}
\end{theorem}

\noindent No independence and no $\sum_j \omega_j = m$ is needed for the displayed bound; $\sum_j \omega_j = m$ gives $\FDR \le q$ when $\delta = 0$.\claim{THM-BY-FIN}

\subsection{Asymptotic false discovery rate}
\label{spec:thm.fdr}

\begin{corollary}[Asymptotic FDR]
If (A1)--(A4) hold for every true sharp null, and the weights and directions are fixed from independent data, then
\begin{equation*}
\limsup_{n \to \infty} \FDR \le \frac{q}{m} \sum_{j \in \Hset} \omega_j \le q
\end{equation*}
whatever the dependence between the p-values.
\end{corollary}

\noindent The step from the finite bound requires that the null p-values satisfy $\limsup_n \sup_t \bigl(F_j(t) - t\bigr) \le 0$ \emph{uniformly in $t$} (so $\delta_n \to 0$ in the theorem above), and weights $\omega_j \ge 0$ with $\sum_j \omega_j = m$ for the bound $q$. This is not formalised.\claim{THM-FDR-ASY}

\subsection{Design slope}
\label{spec:thm.slope}

\begin{theorem}[Design slope]
Assume (A1) with natural history, outcomes bounded by $M$, a predictable centre $\hat m_u$ bounded by $C$, overlap $\pi_u \in [\eta, 1-\eta]$, \emph{episode-level} adaptedness (the increments of an episode $e$ are measurable between $\Fpast{e}$ and $\Fpast{e+1}$), a unit count $N_n \ge \rho n$ with probability tending to one, and the bounded-martingale weak law as a hypothesis. Then $\hat\beta$ converges in probability to the $\pi_u(1-\pi_u)$-weighted average of $\sgn_u \tau_u^\pi$.
\end{theorem}

\noindent Boundedness of the centre is necessary (Lean \texttt{unbounded\_centre\_counterexample}). At $\pi_u = 1/2$ the slope is the Horvitz--Thompson estimator as an identity (\texttt{half\_propensity\_identity}). The paper argues at the unit level; the proof is at the episode level.\claim{THM-SLOPE}

\subsection{Scope of the guarantees}
\label{spec:thm.scope}

The FDR bound covers the declared edge $\hyp$ (that $\Hnull\hyp$ is false), not its sign, relation or magnitude. When $f$ affects its own cells the neighbour and far hypotheses of $f$ are false in the sense of the sharp null and declaring them is correct, even if the effect is only propagated. No proof is given beyond this reading of the definitions.\claim{THM-SCOPE}
